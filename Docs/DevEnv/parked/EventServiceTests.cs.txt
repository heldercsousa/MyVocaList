    [Fact]
    // [AC-2.3, AC-2.4] Event status transitions from CREATED → STARTED; Actual start timestamp is recorded
    public async Task StartEventAsync_EventCreated_TransitionsToStarted()
    {
        var @event = new Event
        {
            Id = 1,
            VenueId = 1,
            Name = "Test Event",
            Status = EventStatus.Created,
            ScheduledStartTime = DateTime.UtcNow.AddHours(1),
            ScheduledEndTime = DateTime.UtcNow.AddHours(2),
            Mode = "VideoKaraoke",
            CreatedAt = DateTime.UtcNow,
            ModifiedAt = DateTime.UtcNow,
            Venue = null!,
            QueueEntries = []
        };
        _repoMock.Setup(r => r.GetByIdAsync(1, It.IsAny<CancellationToken>()))
                 .ReturnsAsync(@event);
        _repoMock.Setup(r => r.UpdateAsync(It.IsAny<Event>(), It.IsAny<CancellationToken>()))
                 .Returns(Task.CompletedTask);

        var (success, message) = await sut.StartEventAsync(1, CancellationToken.None);

        _repoMock.Verify(r => r.UpdateAsync(It.IsAny<Event>(), It.IsAny<CancellationToken>()), Times.Once);
