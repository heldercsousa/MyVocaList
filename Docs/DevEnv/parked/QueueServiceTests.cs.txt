/// <summary>

    // ── EnqueueSingerAsync ─────────────────────────────────────────────────────

    [Fact]
    public async Task EnqueueSingerAsync_ValidInput_ReturnsSuccess()
    {
        var person = new Person { Id = 1, FullName = "João Silva", Email = "joao@example.com" };
        var @event = new Event { Id = 1, Name = "Test Event", Status = EventStatus.Started, VenueId = 1, Venue = null!, QueueEntries = [] };

        _personRepoMock.Setup(r => r.GetByIdAsync(1, It.IsAny<CancellationToken>()))
                       .ReturnsAsync(person);
        _eventRepoMock.Setup(r => r.GetByIdAsync(1, It.IsAny<CancellationToken>()))
                      .ReturnsAsync(@event);
        _queueRepoMock.Setup(r => r.ExistsAsync(1, 1, It.IsAny<CancellationToken>()))
                      .ReturnsAsync(false);
        _queueRepoMock.Setup(r => r.AddAsync(It.IsAny<QueueEntry>(), It.IsAny<CancellationToken>()))
                      .Returns(Task.CompletedTask);

        var sut = CreateSut();

        var (success, message) = await sut.EnqueueSingerAsync(1, 1, null, CancellationToken.None);

        Assert.True(success);
        Assert.NotEmpty(message);
        _queueRepoMock.Verify(r => r.AddAsync(It.IsAny<QueueEntry>(), It.IsAny<CancellationToken>()), Times.Once);
    }

    public async Task EnqueueSingerAsync_PersonNotExists_ReturnsFalse()

    // ── ReorderQueueAsync ──────────────────────────────────────────────────────

    [Fact]
    public async Task ReorderQueueAsync_ValidOrder_UpdatesPositions()
    {
        var entries = new[]
        {
            new QueueEntry { Id = 1, EventId = 1, Position = 0, Status = QueueEntryStatus.Pending, Event = null!, Person = null! },
            new QueueEntry { Id = 2, EventId = 1, Position = 1, Status = QueueEntryStatus.Pending, Event = null!, Person = null! },
            new QueueEntry { Id = 3, EventId = 1, Position = 2, Status = QueueEntryStatus.Pending, Event = null!, Person = null! }
        };

        _queueRepoMock.Setup(r => r.GetByEventIdAsync(1, It.IsAny<CancellationToken>()))
                      .ReturnsAsync(entries);
        _queueRepoMock.Setup(r => r.UpdateAsync(It.IsAny<QueueEntry>(), It.IsAny<CancellationToken>()))
                      .Returns(Task.CompletedTask);

        var sut = CreateSut();
        var newOrder = new[] { (2, 0), (1, 1), (3, 2) };

        var (success, message) = await sut.ReorderQueueAsync(1, newOrder, CancellationToken.None);

        Assert.True(success);
        _queueRepoMock.Verify(r => r.UpdateAsync(It.IsAny<QueueEntry>(), It.IsAny<CancellationToken>()), Times.AtLeast(3));
    }

    [Fact]
    public async Task CalculateCompletionEstimateAsync_5Plus_CalculatesETA()
    {
        var entries = new[]
        {
            new QueueEntry { Id = 1, Status = QueueEntryStatus.Completed, PerformanceDurationMinutes = 3, Event = null!, Person = null! },
            new QueueEntry { Id = 2, Status = QueueEntryStatus.Completed, PerformanceDurationMinutes = 3, Event = null!, Person = null! },
            new QueueEntry { Id = 3, Status = QueueEntryStatus.Completed, PerformanceDurationMinutes = 3, Event = null!, Person = null! },
            new QueueEntry { Id = 4, Status = QueueEntryStatus.Completed, PerformanceDurationMinutes = 3, Event = null!, Person = null! },
            new QueueEntry { Id = 5, Status = QueueEntryStatus.Completed, PerformanceDurationMinutes = 3, Event = null!, Person = null! },
            new QueueEntry { Id = 6, Status = QueueEntryStatus.Pending, Event = null!, Person = null! },
            new QueueEntry { Id = 7, Status = QueueEntryStatus.Pending, Event = null!, Person = null! }
        };

        _queueRepoMock.Setup(r => r.GetByEventIdAsync(1, It.IsAny<CancellationToken>()))
                      .ReturnsAsync(entries);

        var sut = CreateSut();

        var result = await sut.CalculateCompletionEstimateAsync(1, CancellationToken.None);

        Assert.NotNull(result);
        // Average = 3 min, pending = 2, so estimate should be ~6 minutes
        Assert.True(result.Value.TotalMinutes > 0);
    }
